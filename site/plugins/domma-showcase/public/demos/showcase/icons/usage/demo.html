HTML Data Attributes

<!-- Basic usage -->
<span data-icon="check"></span>

<!-- With size -->
<span data-icon="user" data-icon-size="32"></span>

JavaScript API

// Scan and render data-icon attributes
I.scan();

// Create an SVG element
const icon = I.render('heart', {size: 24});
$('#icon-target').append(icon);

// Get an SVG HTML string
const svg = I.html('star', {size: 24, class: 'my-icon'});

// List the available categories
I.listCategories();
