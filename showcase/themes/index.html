Usage

CSS Only (No JavaScript Required)

<!-- Include theme CSS -->
<link rel="stylesheet" href="dist/themes/domma-themes.css">

<!-- Apply theme via class -->
<body class="dm-theme-dark dm-theme-ocean">

JavaScript API

// Quick setup
$.setup({ theme: 'dark', themeVariant: 'forest' });

// Or initialise directly
Domma.theme.init({ theme: 'dark', variant: 'ocean', autoDetect: true });

// Runtime control
Domma.theme.toggle(); // Toggle light/dark
Domma.theme.set('dark'); // Set specific theme
Domma.theme.setVariant('sunset'); // Change colour variant

// Listen for changes
Domma.theme.onChange(({ newTheme, variant }) => {
  console.log('Theme changed:', newTheme, variant);
});

Theme Preview (for Theme Pickers)

// Preview a theme temporarily
const restore = Domma.theme.preview('dark', 'sunset');

// User clicks "Cancel"
restore(); // Reverts to previous theme

// Or user clicks "Apply"
Domma.theme.set('dark');
Domma.theme.setVariant('sunset');
